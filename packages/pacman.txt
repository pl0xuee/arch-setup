# Repo packages that exist everywhere this script runs — CachyOS and plain Arch
# (which is what Omarchy is), on any desktop. One package per line. Blank lines
# and #-comments are ignored. Re-running is safe: pacman --needed skips anything
# already installed.
#
# Anything that ISN'T available everywhere lives in a sibling list instead:
#   pacman-cachyos.txt   only in the CachyOS repos — install.sh adds that repo
#                        on boxes that don't have it, so the list still installs
#   pacman-kde.txt       only worth installing on KDE Plasma

keepassxc                    # Password manager
proton-vpn-gtk-app           # ProtonVPN client
gnome-disk-utility           # Disk management GUI
lact                         # AMD GPU control (fan curves, undervolt)
# freerdp is only an *optional* dep of remmina, so it has to be named
# explicitly — install remmina alone and you get no RDP plugin at all.
remmina                      # Remote desktop client
freerdp                      # Remmina's RDP plugin

# ── Music AI Player ───────────────────────────────────────────────────
# The player itself bundles everything it needs; these two are for its YouTube
# importer, which shells out to whatever is on PATH. Without them the app still
# starts and plays — the Import panel just refuses with "yt-dlp is not
# installed". ffmpeg is what yt-dlp extracts audio with, and ffprobe (same
# package) is what measures each track's real duration for the crossfade.
yt-dlp                       # Music AI Player: YouTube import
ffmpeg                       # Music AI Player: yt-dlp's audio extraction + ffprobe

# ── Gaming: the 32-bit graphics driver ────────────────────────────────
# The 32-bit Vulkan driver for AMD. Nothing else in this setup pulls it in:
# cachyos-gaming-meta doesn't, and lib32-mesa is the 32-bit GL driver — the
# Vulkan ICD is a separate package. It lives in multilib, which is on by default
# on CachyOS and switched on by ensure_cachyos_repo() everywhere else.
#
# Without it, a 32-bit Windows game under Proton dies about three seconds in
# with no error dialog at all: the game's own logs show a CPU probe and then an
# empty gx.log, and DXVK's x86 build reports "Failed to create Vulkan instance"
# — because the only ICD on the box (/usr/share/vulkan/icd.d/radeon_icd.json)
# is the 64-bit one. The 64-bit half of the same Proton session enumerates the
# GPU perfectly, which is what makes it look like the game, not the driver.
#
# WoW 3.3.5a is a 32-bit exe, and so is most of what runs against a private
# server. CachyOS ships this driver as part of its own install, so the gap only
# appears after moving to a plain-Arch box like Omarchy.
lib32-vulkan-radeon          # 32-bit RADV — every 32-bit game under Proton needs it
